let stock_master = datatable(symbol:string, companyName:string, exchange:string, country:string)
[
   "NSFT", "Nosoft Technologies", "NASDAQ", "USA",
   "BMZM", "BizMart Retail", "NYSE", "USA",
   "HOOJ", "Hooj Media", "NASDAQ", "USA"
];
stocks
| summarize
   AvgPrice = avg(lastSalePrice),
   TotalVolume = sum(volume),
   RecordCount = count()
 by symbol
| join kind=inner stock_master on symbol
| project symbol, companyName, exchange, AvgPrice, TotalVolume, RecordCount
| order by TotalVolume desc 

//time Series
stocks
| summarize RecordCount = count() by bin(['time'], 1m)
| order by ['time'] asc

stocks
| summarize AvgPrice = avg(lastSalePrice) by bin(['time'], 1m)
| order by ['time'] asc

.create materialized-view Stocks_1Min_Summary on table stocks
{
   stocks
   | summarize
       AvgPrice = avg(lastSalePrice),
       TotalVolume = sum(volume),
       RecordCount = count()
     by bin(['time'], 1m), symbol
}
Stocks_1Min_Summary
| order by ['time'] desc
| take 100

Stocks_1Min_Summary

| where symbol == "NSFT"

| order by ['time'] asc

.show materialized-view Stocks_1Min_Summary
